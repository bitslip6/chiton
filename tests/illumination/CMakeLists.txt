cmake_minimum_required(VERSION 3.10)
project(illumination_fixtures LANGUAGES CXX)
if(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Illumination fixtures are host-only; do not execute ARM tests")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
get_filename_component(CHITON_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(case illumination_controller illumination_register_adapter illumination_camera_policy)
  add_executable(${case}_test "${CHITON_SOURCE}/tests/${case}.cpp"
    "${CHITON_SOURCE}/src/graphics/illumination_controller.cpp")
  target_include_directories(${case}_test PRIVATE "${CHITON_SOURCE}/include")
  target_compile_options(${case}_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  target_link_libraries(${case}_test PRIVATE Threads::Threads)
  add_test(NAME ${case} COMMAND ${case}_test)
  set_tests_properties(${case} PROPERTIES TIMEOUT 120)
endforeach()
target_sources(illumination_camera_policy_test PRIVATE "${CHITON_SOURCE}/src/system/illumination_sdk.cpp")
target_compile_definitions(illumination_camera_policy_test PRIVATE CHITON_EXPERIMENTAL_ILLUMINATION=1)
